import Anthropic from "@anthropic-ai/sdk";
import type { BetaManagedAgentsSessionEvent as Event } from "@anthropic-ai/sdk/resources/beta/sessions/events";
type Ask = Extract<Event, { type: "agent.tool_use" | "agent.mcp_tool_use" }>;
export type Decision = { result: "allow" } | { result: "deny"; denyMessage: string };
type State = "reviewing" | "sending" | "sent" | "uncertain";
export function makeConfirmer(client: Anthropic, sessionId: string,
  decide: (call: Ask) => Promise<Decision>) {
  // One instance per session and one serialized owner. Persist claims/decisions in production.
  const states = new Map<string, State>();
  return { states, async onEvent(event: Event) {
    if (event.type !== "agent.tool_use" && event.type !== "agent.mcp_tool_use") return;
    if (event.evaluated_permission !== "ask" || states.has(event.id)) return;
    states.set(event.id, "reviewing"); // claim before awaiting a person/policy
    let decision: Decision;
    try { decision = await decide(event); }
    catch { decision = { result: "deny", denyMessage: "Review unavailable or expired." }; }
    states.set(event.id, "sending");
    try {
      await client.beta.sessions.events.send(sessionId, { events: [{
        type: "user.tool_confirmation", tool_use_id: event.id, result: decision.result,
        ...(decision.result === "deny" ? { deny_message: decision.denyMessage } : {})
      }] }, { maxRetries: 0, timeout: 10000 });
      states.set(event.id, "sent");
    } catch { states.set(event.id, "uncertain"); }
  } };
}
// decide must authenticate the reviewer, authorize the session and bound review duration.
// It displays the original name/input and returns a decision; no input rewriting is supported.
// On reconnect, reconcile latest pending calls/history and durable decision state first.
// An uncertain acknowledgement is not proof of rejection: do not blindly resubmit.
// sevt_ correlates a call. It is not a stable business idempotency key for a side effect.
